function listen(){
  unsub.forEach(f=>f());unsub=[];
  const P=DB.collection('finanzas').doc(ROOT);
  const all=SY.rol==='admin'||isReadOnly();
  const make=(col)=>all?P.collection(col):P.collection(col).where('uid','==',SY.uid);
  const merge=(target,base,q,mapFn,col)=>{
    const merged=new Map(base.map(x=>[x.id,{...x}]));
    q.docs.map(mapFn).forEach(x=>{
      // V14: los históricos se toman exclusivamente de los Excel originales reconstruidos.
      // Firestore complementa solo movimientos nuevos creados desde la plataforma.
      if(x.historico===true || String(x.id||'').startsWith('hist_') || String(x.id||'').startsWith('tenjo_') || String(x.id||'').startsWith('madrid_')) return;
      merged.set(x.id,x);
    });
    target.splice(0,target.length,...Array.from(merged.values()).sort(sortTs));
    USING_EMBEDDED_FALLBACK=true;
    render();
  };
  const fail=(col,e)=>{console.error('Firestore '+col,e);toast('Firebase no pudo leer '+col+'. El Dashboard conserva los datos de los Excel originales.');render()};
  unsub.push(make('gastos').onSnapshot(q=>merge(G,EMBED_G,q,d=>({id:d.id,...d.data()}),'gastos'),e=>fail('gastos',e)));
  if(all){unsub.push(P.collection('ingresos').onSnapshot(q=>merge(I,EMBED_I,q,d=>({id:d.id,...d.data()}),'ingresos'),e=>fail('ingresos',e)))}else I=[...EMBED_I];
  unsub.push(make('prestamos').onSnapshot(q=>merge(L,EMBED_L,q,d=>({id:d.id,...d.data()}),'prestamos'),e=>fail('préstamos',e)));
  unsub.push(make('deudas_tecnicos').onSnapshot(q=>merge(D,EMBED_D,q,d=>({id:d.id,...d.data()}),'deudas'),e=>fail('deudas',e)));
  unsub.push(make('nomina').onSnapshot(q=>merge(N,EMBED_N,q,d=>({id:d.id,...d.data()}),'nomina'),e=>fail('nómina',e)));
}
