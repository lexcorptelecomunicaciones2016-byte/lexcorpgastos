  allow read: if request.auth != null;
  allow create, update, delete: if admin();
}

  allow read: if admin() ||
    (activo() && resource.data.uid == request.auth.uid);

  allow create: if activo() &&
    request.resource.data.uid == request.auth.uid;

  // SOLO ADMON puede corregir o eliminar movimientos contables.
  allow update, delete: if admin();
}
